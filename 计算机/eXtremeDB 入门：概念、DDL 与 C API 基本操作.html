eXtremeDB 入门：概念、DDL 与 C API 基本操作 ​

eXtremeDB 是 McObject 公司的内存嵌入式数据库，面向高性能、实时性敏感的场景，提供 C/C++ 接口与符合 ACID 的事务。它的使用方式与常见的关系数据库差异较大：schema 用专门的 DDL 语言声明，编译成字典后由一组 mco_* 前缀的 C 函数驱动。本文整理入门所需的三个部分：核心概念、DDL 写法、C API 的完整操作流程，并附一份可直接对照官方 samples 阅读的完整示例。适合需要在嵌入式或实时项目中引入内存数据库的 C/C++ 开发者。

1 核心概念 ​

eXtremeDB 用一套自己的术语组织数据，与关系数据库的概念大致对应但并不等同：

eXtremeDB 关系数据库对应物 说明
class table 一组元素的集合
instance record 类的实例化对象
field 列（属性） 类中有类型的属性
structure —— 复杂类型，作为 class 的组成部分，不能被实例化

与面向对象语言类似，eXtremeDB 中的数据库类定义了对象的字段和字段的访问方法。字段支持的基础类型（uint1、int4、char<N>、string、blob 等）见官方文档 Base Data Types。

oid 与 ref：比索引更适合建立类间关系 ​

eXtremeDB 提供两种对象寻址机制：

  • oid（object identifier）：对象标识符。除了索引之外，类可以声明一个 oid；oid 的组成对数据库中的每一个类都是相同的，是数据库范围内统一的对象编号。
  • ref（引用）：一个类中数据类型为 ref 的字段，存放对另一个类对象的 oid 引用，类似指针。vector<ref> 可以建立一对多关系。

主键通常实现为类中字段或字段组合的唯一索引，外键通常实现为非唯一索引（允许重复），其元素对应其他类的主键。但建立类间关系时，oid 和 ref 是比 index 更直接的方式——index 的本职是加速搜索，不应承担关系语义。

两个零散但重要的点 ​

  1. 数据库字典：schema 经 DDL 编译后的二进制文件，打开数据库时传入（即后文 mco_db_open_dev 的 dict 参数）。
  2. 隐式 list 索引：在没有声明其他索引的情况下，类会被隐式创建一个 list 索引，使对象可以被顺序扫描。

2 DDL：声明 schema ​

DDL（Data Definition Language）是 eXtremeDB 的 schema 声明语言，独立于 C/C++ 源码，编译后生成数据库字典。下面这个例子覆盖了常用语法：

c
// 预处理：类型别名，DDL 原生写法是 signed<1> / unsigned<2> 等
#define int1 signed<1>
#define int2 signed<2>
#define int4 signed<4>
#define uint4 unsigned<4>
#define uint2 unsigned<2>
#define uint1 unsigned<1>

// 结构体声明：作为 class 的组成部分，不能被实例化
struct SampleStruct {
    uint2 s1;
    char<20> s2;
};
struct BigStruct {
    string      str;
    uint2       u2;
    uint4       u4;
    vector<SampleStruct> vss;
};

// DDL 声明语句：声明数据库名
declare database simple;

// 为 SampleStruct 声明 oid 索引，方括号内为预估的实例数（容量提示）
declare OID SampleStruct[20000];

// compact：总对象大小（含开销）小于 64K，大小计算不计 blob 字段
compact class SampleClass {

    /* 基本数据类型 */
    uint1 a = 0;
    uint2 b;
    uint4 c;

    /* oid 引用字段 */
    ref   d;

    /* 向量：元素可以是任意类型 */
    vector<uint2>        numbers;
    vector<SampleStruct> ss;

    /* string 上限 64K */
    string str;

    /* 二进制大对象，无大小限制 */
    blob   blo;

    /* optional 结构体：值可以缺失 */
    optional BigStruct big_struct;

    /* 索引声明 */
    voluntary unique tree<a, b, ss.s2> SAM;
    hash<a>     AAAA[10000];
    hash<ss.s2> SSS2[10000];
    hash<numbers> NNNN[10000];

    /* 事件（触发器） */
    event<new>       new;
    event<delete>    delete;
    event<a update>  update;

    autoid;   // 自动递增 id
    oid;      // 对象标识符
    list;     // 顺序扫描索引
};

几处语法的语义：

  • compact class：对象紧凑存放，总大小（含数据库开销）小于 64K 时可用，大小计算不计嵌入的 blob 字段。
  • 索引关键字：tree 为树索引（支持排序顺序），hash 为哈希索引，list 表示对象可顺序扫描；索引可带方括号内的预估实例数。字段名可以用点号嵌套引用结构体成员（如 ss.s2）。
  • voluntary：索引可在运行时创建和删除。unique：唯一索引。
  • event：在对象新建、删除或指定字段更新时触发的事件（触发器机制）。
  • autoid 与 oid：前者是类内自动递增编号，后者是数据库范围统一的对象标识；一个类只能有一个 oid 声明。

完整语法见官方文档 A DDL Example。

3 C API 基本操作 ​

生命周期：从挂载到卸载的 11 步 ​

一个完整的 eXtremeDB 程序遵循固定的生命周期，顺序不能乱：

  1. 挂载文件系统：sample_os_initialize(DEFAULT);（sample 框架封装，内部调用 mount()）
  2. 启动运行环境：mco_runtime_start();
  3. 打开数据库：mco_db_open_dev(dbname, dict, devs, n_devs, params)
  4. 连接数据库：mco_db_connect(dbname, &db)
  5. 开启事务：mco_trans_start(db, trans_type, priority, &t)
  6. CRUD：实例化对象、字段读写（见下文命名规则）
  7. 提交事务：mco_trans_commit(t)
  8. 断开连接：mco_db_disconnect(db)
  9. 关闭数据库：mco_db_close(dbname)
  10. 停止运行环境：mco_runtime_stop()
  11. 卸载文件系统：sample_os_shutdown();（内部调用 umount()）

其中事务相关的两个原型：

c
MCO_RET mco_trans_start(/*IN*/ mco_db_h db,              // 连接句柄
                        /*IN*/ MCO_TRANS_TYPE trans_type,     // 事务类型：MCO_READ_WRITE / MCO_READ_ONLY
                        /*IN*/ MCO_TRANS_PRIORITY priority,   // 事务优先级：如 MCO_TRANS_FOREGROUND
                        /*OUT*/ mco_trans_h *t);              // 事务句柄

CRUD 的命名规则 ​

DDL 中的每个类和字段，都会生成一组按 classname_method_fieldname 规则命名的 C 函数，通过事务句柄操作对象：

c
/* 实例化对象 */
A_new(t, &a);
/* 标量字段赋值 */
A_ui1_put(&a, 1);
/* 字符串字段赋值（显式长度） */
A_s_put(&a, buf, (uint2)strlen(buf));
/* 读取字段 */
A_ui1_get(&a, &ui1);
/* 获取结构体字段的写句柄，再操作结构体内部 */
A_b_write_handle(&a, &b);
B_i1_put(&b, 4);
/* 遍历：为类创建列表游标，移动游标并取回对象 */
A_list_cursor(t, &csr);
mco_cursor_first(t, &csr);
A_from_cursor(t, &csr, &a);

完整示例 ​

下面的例子来自官方 samples 的典型结构：写入一个带各类型字段与嵌套结构体的对象，提交后重新开只读事务读回并打印。sample_open_database 是 sample 框架对 mco_db_open_dev 的封装（处理设备、页大小、最大连接数等参数）：

c
#include <stdio.h>
#include <common.h>
#include <simpledb.h>

const char *db_name = "operations";

int main(int argc, char *argv[])
{
    MCO_RET rc;
    sample_memory_t dbmem;

    /* 1. 挂载文件系统 */
    sample_os_initialize(DEFAULT);

    /* 2. 启动 eXtremeDB 运行环境 */
    mco_runtime_start();

    /* 3. 打开数据库（sample 框架封装：数据库名、字典、库与 cache 大小、页大小、最大连接数） */
    rc = sample_open_database(db_name, simpledb_get_dictionary(), DATABASE_SIZE, CACHE_SIZE,
                              MEMORY_PAGE_SIZE, PSTORAGE_PAGE_SIZE, 1, &dbmem);
    sample_rc_check("\tOpen", rc);

    if (MCO_S_OK == rc)
    {
        mco_db_h connection; /* 连接句柄 */

        /* 4. 连接数据库 */
        rc = mco_db_connect(db_name, &connection);
        sample_rc_check("\tConnect", rc);

        if (MCO_S_OK == rc)
        {
            mco_trans_h t; /* 事务句柄 */

            /* 5. 开启读写事务 */
            rc = mco_trans_start(connection, MCO_READ_WRITE, MCO_TRANS_FOREGROUND, &t);
            if (MCO_S_OK == rc)
            {
                A a;         /* class A 的实例句柄 */
                A_fixed _a;  /* A 的定长部分快照 */
                B b;         /* 结构体 B 的句柄 */
                B_fixed _b;
                uint1 ui1; uint2 ui2; uint4 ui4;
                int1 i1;  int2 i2;  int4 i4;
                mco_int8 i8;
                float f = (float)1.1;
                double d = 2.2;
                char buf[16];
                mco_cursor_t csr;

                /* 6a. 写入：实例化并填充各类型字段 */
                A_new(t, &a);
                A_ui1_put(&a, 1);
                A_ui2_put(&a, 2);
                A_ui4_put(&a, 3);
                A_d_put(&a, d);
                A_s_put(&a, db_name, (uint2)strlen(db_name)); /* 字符串字段 */

                A_num_from_chars(&i8, "1234567");  /* 数字字段支持字符串转换 */
                A_num_put(&a, i8);
                A_dec_to_chars(987654321, buf, sizeof(buf)); /* decimal 字段同理 */
                A_dec_put_chars(&a, buf);

                /* 嵌套结构体：先取写句柄，再写内部字段 */
                A_b_write_handle(&a, &b);
                B_i1_put(&b, 4);
                B_i2_put(&b, 5);
                B_i4_put(&b, 6);
                B_f_put(&b, f);
                B_c10_put(&b, db_name, (uint2)strlen(db_name));

                /* 7. 提交事务 */
                rc = mco_trans_commit(t);
                sample_rc_check("\tCommit Transaction", rc);

                /* 6b. 读取：开只读事务，用游标定位对象 */
                rc = mco_trans_start(connection, MCO_READ_ONLY, MCO_TRANS_FOREGROUND, &t);
                if (MCO_S_OK == rc)
                {
                    rc = A_list_cursor(t, &csr);   /* 列表游标（无序导航） */
                    if (MCO_S_OK == rc)
                    {
                        mco_cursor_first(t, &csr);
                        A_from_cursor(t, &csr, &a);

                        A_ui1_get(&a, &ui1);
                        A_ui2_get(&a, &ui2);
                        A_ui4_get(&a, &ui4);
                        A_d_get(&a, &d);
                        printf("ui1=%d, ui2=%d, ui4=%ld, d=%f\n", ui1, ui2, ui4, d);

                        A_num_get(&a, &i8);
                        A_num_get_chars(&a, buf, sizeof(buf));
                        printf("num=%lld, chars(%s)\n", i8, buf);

                        A_dec_get(&a, &i8);
                        A_dec_get_chars(&a, buf, sizeof(buf));
                        printf("dec=%lld, chars(%s)\n", i8, buf);

                        /* 注意：缓冲区足够长时 _get 才会补 '\0' 终止符 */
                        A_s_get(&a, buf, sizeof(buf), &ui2);
                        printf("string s=(%s), (length = %d)\n", buf, ui2);

                        /* 嵌套结构体：取读句柄后读内部字段 */
                        A_b_read_handle(&a, &b);
                        B_i1_get(&b, &i1);
                        B_i2_get(&b, &i2);
                        B_i4_get(&b, &i4);
                        B_f_get(&b, &f);
                        B_c10_get(&b, buf, sizeof(buf));
                        printf("Struct b: i1=%d, i2=%d, i4=%ld, f=%f, c10=(%s)\n",
                               i1, i2, i4, f, buf);

                        /* 定长快照方式：一次取回对象的定长部分 */
                        A_fixed_get(&a, &_a);
                        printf("Using A_fixed: ui1=%d, ui2=%d, ui4=%ld, d=%f\n",
                               _a.ui1, _a.ui2, _a.ui4, _a.d);

                        B_fixed_get(&b, &_b);
                        strncpy(buf, _b.c10, 10); /* 定长字符数组需手动补终止符 */
                        buf[10] = '\0';
                        printf("Using B_fixed: i1=%d, i2=%d, i4=%ld, f=%f, c10=(%s)\n",
                               _b.i1, _b.i2, _b.i4, _b.f, buf);
                    }
                    mco_trans_commit(t); /* 只读事务同样要提交（释放资源） */
                }
            }

            /* 8. 断开连接 */
            rc = mco_db_disconnect(connection);
            sample_rc_check("\tDisconnect", rc);
        }

        /* 9. 关闭数据库 */
        sample_close_database(db_name, &dbmem);
        sample_rc_check("\tClose", rc);
    }

    /* 10. 停止 eXtremeDB 运行环境 */
    mco_runtime_stop();

    sample_pause_end("\n\nPress any key to continue . . . ");

    /* 11. 卸载文件系统 */
    sample_os_shutdown();

    return 0;
}

4 要点 ​

  • 术语映射：class/instance/field 对应 table/record/列；structure 不能实例化，只作 class 的组成部分。
  • 关系建模优先用 oid 与 ref（vector<ref> 建一对多），索引（tree/hash/list）留给搜索加速；无其他索引时隐式创建 list 索引。
  • DDL 声明 schema 并编译成数据库字典，类与字段生成 classname_method_fieldname 命名的 C 函数，所有对象操作都挂在事务句柄上。
  • 程序生命周期 11 步，顺序固定：挂载文件系统 → 启动 runtime → 打开库 → 连接 → 事务内 CRUD → 提交 → 断开 → 关库 → 停 runtime → 卸载。
  • 只读事务结束同样要调用 mco_trans_commit 释放资源；字符串 _get 只在缓冲区足够长时才补 \0。

官方文档与发行版内的 samples 是最可靠的进阶材料；如果在 ARM 交叉编译环境中构建 eXtremeDB 本体，常见的编译失败可参考《eXtremeDB ARM 构建失败排查》一文。
